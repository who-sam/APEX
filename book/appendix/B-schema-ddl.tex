\chapter{Schema DDL}
\label{appendix:schema-ddl}

This appendix documents the durable schema. The canonical source is the GORM model files under \texttt{internal/models/} plus the idempotent SQL in \texttt{internal/database/migrations.go}; the listing below is what \texttt{pg\_dump --schema-only --no-owner --no-privileges} produces against a fresh deployment, lightly elided for readability.

\section{Core tables}

\begin{lstlisting}[language=SQL,basicstyle=\ttfamily\footnotesize,breaklines=true]
CREATE TABLE users (
    id              BIGSERIAL PRIMARY KEY,
    email           VARCHAR(255) NOT NULL UNIQUE,
    password_hash   TEXT NOT NULL,
    role            VARCHAR(20) NOT NULL,
    name            VARCHAR(255) NOT NULL,
    google_id       VARCHAR(64),
    created_at      TIMESTAMPTZ DEFAULT now()
);
CREATE UNIQUE INDEX idx_users_google_id ON users(google_id)
    WHERE google_id IS NOT NULL AND google_id <> '';

CREATE TABLE user_profiles (
    id                            BIGSERIAL PRIMARY KEY,
    user_id                       BIGINT UNIQUE,
    bio                           TEXT,
    avatar_url                    TEXT,
    notify_email                  BOOLEAN DEFAULT true,
    notify_push                   BOOLEAN DEFAULT true,
    notify_exam_reminders         BOOLEAN DEFAULT true,
    notify_results                BOOLEAN DEFAULT false,
    notify_exam_email             BOOLEAN NOT NULL DEFAULT false,
    block_announce_with_pending   BOOLEAN NOT NULL DEFAULT true,
    default_exam_draft            BOOLEAN NOT NULL DEFAULT true,
    default_passing_threshold     INTEGER NOT NULL DEFAULT 60
);

CREATE TABLE password_reset_tokens (
    id          BIGSERIAL PRIMARY KEY,
    user_id     BIGINT NOT NULL,
    token_hash  VARCHAR(64) NOT NULL UNIQUE,
    expires_at  TIMESTAMPTZ NOT NULL,
    used_at     TIMESTAMPTZ,
    created_at  TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE classes (
    id                            BIGSERIAL PRIMARY KEY,
    teacher_id                    BIGINT NOT NULL,
    name                          VARCHAR(255) NOT NULL,
    section                       VARCHAR(100),
    invite_code                   VARCHAR(8) NOT NULL UNIQUE,
    cover_image                   TEXT,
    grades_announced              BOOLEAN NOT NULL DEFAULT false,
    passing_threshold             INTEGER NOT NULL DEFAULT 60,
    block_announce_with_pending   BOOLEAN NOT NULL DEFAULT true,
    created_at                    TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE class_members (
    id        BIGSERIAL PRIMARY KEY,
    class_id  BIGINT NOT NULL,
    user_id   BIGINT NOT NULL,
    joined_at TIMESTAMPTZ DEFAULT now(),
    UNIQUE (class_id, user_id)
);

CREATE TABLE folders (
    id          BIGSERIAL PRIMARY KEY,
    teacher_id  BIGINT NOT NULL,
    name        VARCHAR(255) NOT NULL,
    created_at  TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE announcements (
    id           BIGSERIAL PRIMARY KEY,
    class_id     BIGINT NOT NULL,
    teacher_id   BIGINT NOT NULL,
    title        VARCHAR(255) NOT NULL,
    body         TEXT,
    attachments  JSONB DEFAULT '[]',
    created_at   TIMESTAMPTZ DEFAULT now()
);
\end{lstlisting}

\section{Assessment tables}

\begin{lstlisting}[language=SQL,basicstyle=\ttfamily\footnotesize,breaklines=true]
CREATE TABLE exams (
    id                             BIGSERIAL PRIMARY KEY,
    teacher_id                     BIGINT NOT NULL,
    title                          VARCHAR(255) NOT NULL,
    description                    TEXT,
    duration_minutes               INTEGER NOT NULL DEFAULT 60,
    start_time                     TIMESTAMPTZ,
    end_time                       TIMESTAMPTZ,
    reset_at                       TIMESTAMPTZ,
    reminder_sent_at               TIMESTAMPTZ,
    email_reminder1h_sent_at       TIMESTAMPTZ,
    email_reminder_start_sent_at   TIMESTAMPTZ,
    created_at                     TIMESTAMPTZ DEFAULT now(),
    shuffle_questions              BOOLEAN DEFAULT false,
    show_results_after             BOOLEAN DEFAULT true,
    passing_score                  INTEGER DEFAULT 50,
    is_practice                    BOOLEAN DEFAULT false,
    is_draft                       BOOLEAN DEFAULT false
);
CREATE INDEX idx_exams_reset_at ON exams(reset_at);

CREATE TABLE exam_classes (
    id        BIGSERIAL PRIMARY KEY,
    exam_id   BIGINT NOT NULL,
    class_id  BIGINT NOT NULL,
    UNIQUE (exam_id, class_id)
);

CREATE TABLE problems (
    id                       BIGSERIAL PRIMARY KEY,
    exam_id                  BIGINT,
    is_bank                  BOOLEAN NOT NULL DEFAULT false,
    teacher_id               BIGINT,
    class_id                 BIGINT,
    folder_id                BIGINT REFERENCES folders(id) ON DELETE SET NULL,
    title                    VARCHAR(255) NOT NULL,
    description              TEXT NOT NULL,
    type                     VARCHAR(20) NOT NULL DEFAULT 'coding',
    points                   INTEGER NOT NULL DEFAULT 10,
    difficulty               VARCHAR(20) NOT NULL DEFAULT 'medium',
    starter_code             TEXT,
    hints                    TEXT,
    time_limit_ms            INTEGER NOT NULL DEFAULT 2000,
    memory_limit_kb          INTEGER NOT NULL DEFAULT 262144,
    order_index              INTEGER NOT NULL DEFAULT 0,
    options                  JSONB,
    correct_option_ids       JSONB,
    multiple_correct         BOOLEAN DEFAULT false,
    explanation              TEXT,
    max_word_count           INTEGER DEFAULT 500,
    rubric                   TEXT,
    require_manual_grading   BOOLEAN DEFAULT false,
    image_url                TEXT,
    tags                     JSONB DEFAULT '[]',
    created_at               TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE test_cases (
    id              BIGSERIAL PRIMARY KEY,
    problem_id      BIGINT NOT NULL,
    input           TEXT NOT NULL,
    expected_output TEXT NOT NULL,
    is_sample       BOOLEAN NOT NULL DEFAULT false,
    points          INTEGER NOT NULL DEFAULT 0,
    order_index     INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE exam_attempts (
    id              BIGSERIAL PRIMARY KEY,
    user_id         BIGINT NOT NULL,
    exam_id         BIGINT NOT NULL,
    started_at      TIMESTAMPTZ DEFAULT now(),
    submitted_at    TIMESTAMPTZ,
    score           DOUBLE PRECISION NOT NULL DEFAULT 0,
    status          VARCHAR(30) NOT NULL DEFAULT 'in_progress',
    graded_notified BOOLEAN NOT NULL DEFAULT false,
    draft_answers   JSONB,
    draft_saved_at  TIMESTAMPTZ,
    UNIQUE (user_id, exam_id)
);

CREATE TABLE submissions (
    id                 BIGSERIAL PRIMARY KEY,
    user_id            BIGINT NOT NULL,
    problem_id         BIGINT NOT NULL,
    exam_id            BIGINT NOT NULL,
    exam_attempt_id    BIGINT,
    type               VARCHAR(20) NOT NULL DEFAULT 'coding',
    language           VARCHAR(20),
    code               TEXT,
    selected_options   JSONB,
    text_answer        TEXT,
    status             VARCHAR(30) NOT NULL DEFAULT 'pending',
    teacher_feedback   TEXT,
    passed_count       INTEGER NOT NULL DEFAULT 0,
    total_count        INTEGER NOT NULL DEFAULT 0,
    score              DOUBLE PRECISION NOT NULL DEFAULT 0,
    execution_time_ms  INTEGER DEFAULT 0,
    memory_kb          INTEGER DEFAULT 0,
    submitted_at       TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE test_results (
    id                BIGSERIAL PRIMARY KEY,
    submission_id     BIGINT NOT NULL,
    test_case_id      BIGINT NOT NULL,
    passed            BOOLEAN NOT NULL DEFAULT false,
    actual_output     TEXT,
    execution_time_ms INTEGER DEFAULT 0,
    memory_kb         INTEGER DEFAULT 0,
    status            VARCHAR(30)
);

CREATE TABLE notifications (
    id          BIGSERIAL PRIMARY KEY,
    user_id     BIGINT NOT NULL,
    type        VARCHAR(20) NOT NULL,
    title       VARCHAR(255) NOT NULL,
    description TEXT,
    link_to     VARCHAR(255),
    read        BOOLEAN DEFAULT false,
    created_at  TIMESTAMPTZ DEFAULT now()
);
\end{lstlisting}

\section{Indexes worth noting}

\begin{itemize}
  \item \texttt{exam\_attempts (user\_id, exam\_id)} unique --- enforces one attempt per student per exam.
  \item \texttt{exam\_classes (exam\_id, class\_id)} unique --- enforces one assignment row per pair.
  \item \texttt{class\_members (class\_id, user\_id)} unique --- prevents duplicate enrolment.
  \item \texttt{users.google\_id} partial unique (only when present) --- supports linking by Google subject.
  \item \texttt{exams.reset\_at} indexed --- supports the cache-bust query in the take screen.
\end{itemize}
